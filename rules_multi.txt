# Задача 2: условия помещения -> капризность растения
VAR light IN 0 100 "Освещённость помещения"
VAR time IN 0 120 "Время на уход (мин/нед)"
VAR temp IN 10 30 "Температура (°C)"
VAR capr OUT 0 10 "Капризность растения"

SET light low trap 0 0 20 40
SET light mid tri 25 50 75
SET light high trap 60 80 100 100
SET time little trap 0 0 15 45
SET time medium tri 30 60 90
SET time much trap 75 105 120 120
SET temp cool trap 10 10 14 19
SET temp warm tri 17 22 27
SET temp hot trap 25 28 30 30
SET capr very_easy trap 0 0 1 3
SET capr easy tri 1 3 5
SET capr medium tri 3 5 7
SET capr hard tri 5 7 9
SET capr very_hard trap 7 9 10 10

RULE IF light IS low AND time IS little THEN capr IS very_easy
RULE IF light IS low AND time IS much THEN capr IS easy
RULE IF light IS mid AND time IS little THEN capr IS easy
RULE IF light IS mid AND time IS medium THEN capr IS medium
RULE IF light IS mid AND time IS much AND temp IS warm THEN capr IS hard
RULE IF light IS high AND time IS little AND temp IS hot THEN capr IS very_easy
RULE IF light IS high AND time IS medium THEN capr IS medium
RULE IF light IS high AND time IS much AND temp IS warm THEN capr IS very_hard
RULE IF temp IS cool AND time IS little THEN capr IS easy

PLANT 1 "Замиокулькас"
PLANT 3 "Кактус"
PLANT 5 "Спатифиллум"
PLANT 7 "Фикус Бенджамина"
PLANT 9 "Орхидея"

CASE light=15 time=10 temp=20
CASE light=50 time=60 temp=22
CASE light=50 time=110 temp=24
CASE light=85 time=10 temp=28
CASE light=85 time=100 temp=24
CASE light=30 time=5 temp=12
